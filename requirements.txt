# Runtime dependencies for PySimplePMT.
#
# Everything listed here is bundled into the packaged build by PyInstaller,
# so an end user installing the .deb needs none of it - not even Python.
# See packaging/README.md for what the built package does still rely on.
#
# Upper bounds are deliberate: a packaged desktop app should be rebuilt and
# retested against a new major version rather than silently picking one up.

# --- Direct dependencies -------------------------------------------------

customtkinter>=5.2.0,<7.0.0     # UI toolkit
plotly>=5.0.0,<7.0.0           # Interactive Gantt chart rendering
numpy>=1.21.0,<3.0.0            # Required by plotly
openpyxl>=3.0.0,<4.0.0          # XLSX import and export
tkinterweb>=4.0.0,<5.0.0       # Required for embedding Plotly charts in Tkinter

# Enhanced drag-and-drop. The task list falls back to plain Tkinter bindings
# when this is missing, but the packaged build ships with it.
tkinterdnd2>=0.3.0,<0.5.0

# --- Transitive dependencies ---------------------------------------------
# Listed explicitly so the packaged set is auditable and reproducible rather
# than whatever the resolver happens to choose on build day.

darkdetect>=0.7.0,<1.0.0        # customtkinter: system theme detection
packaging>=20.0                 # customtkinter, plotly
python-dateutil>=2.7,<3.0.0     # plotly: date handling
pytz>=2020.1                    # plotly: timezone support (optional)
six>=1.5                        # python-dateutil
et-xmlfile>=1.1.0,<3.0.0        # openpyxl
pandoc>=2.0.0                   # plotly: HTML conversion (optional)
pygments>=2.4.0                 # plotly: syntax highlighting (optional)

# --- Deliberately NOT bundled --------------------------------------------
#
# MS Project (.mpp) import stays optional. Install the reader yourself if you
# need it from a source checkout:
#
